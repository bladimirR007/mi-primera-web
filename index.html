packages = ["openpyxl"] import openpyxl from io import BytesIO from js import document, Uint8Array, window, console import json # ---------- Variables globales ---------- resultado_bytes = None resumen_filtro1 = [] resumen_filtro2 = [] # ---------- Utilidades ---------- def log(msg): """Escribe en la consola del navegador.""" console.log(msg) def set_status(msg, tipo="ok"): el = document.getElementById("status") el.innerText = msg el.className = "status " + tipo def set_output(msg): document.getElementById("output").innerText = msg def mostrar_loading(show): document.getElementById("loading").style.display = "block" if show else "none" # ---------- Carga del archivo ---------- async def leer_archivo(): file_input = document.getElementById("file-input") if not file_input.files.length: return None file = file_input.files.item(0) buffer = await file.arrayBuffer() data = Uint8Array.new(buffer).to_py() return BytesIO(bytes(data)) # ---------- Lógica principal ---------- async def procesar_archivo(event=None): global resultado_bytes, resumen_filtro1, resumen_filtro2 mostrar_loading(True) set_status("⏳ Procesando archivo...", "loading") set_output("Trabajando...") document.getElementById("download-btn").style.display = "none" document.getElementById("stats").style.display = "none" try: bytes_io = await leer_archivo() if bytes_io is None: set_status("⚠️ Selecciona primero un archivo Excel.", "error") set_output("") mostrar_loading(False) return # ========== CARGA DEL EXCEL ========== wb = openpyxl.load_workbook(bytes_io) ws = wb.active # Definir filtros y columnas deseadas columna_filtro1 = "TRAMITE CERRADO/ABIERTO" valor_filtro1 = "Tramite en proceso" columna_filtro2 = "OBSERVACION" valor_filtro2 = "URGENCIA VITAL" columna_filtro3 = "TIPO DE TRAMITE" valor_filtro3 = "REMISION" columnas_deseadas = ["NOMBRE", "APELLIDO", "FUNCIONARIO"] # Identificar índices de columnas indices = {} filtro_index1 = None filtro_index2 = None filtro_index3 = None for i, cell in enumerate(ws[1], start=1): if cell.value == columna_filtro1: filtro_index1 = i if cell.value == columna_filtro2: filtro_index2 = i if cell.value == columna_filtro3: filtro_index3 = i if cell.value in columnas_deseadas: indices[cell.value] = i # Validaciones if filtro_index1 is None: raise ValueError(f"No se encontró la columna '{columna_filtro1}'") if filtro_index2 is None: raise ValueError(f"No se encontró la columna '{columna_filtro2}'") if filtro_index3 is None: raise ValueError(f"No se encontró la columna '{columna_filtro3}'") log(f"Columnas encontradas: {indices}") # ---------- Filtro 1: Tramite en proceso + REMISION ---------- resumen_filtro1 = [] contador1 = 0 for row in ws.iter_rows(min_row=2, values_only=True): if (row[filtro_index1 - 1] and str(row[filtro_index1 - 1]).strip().upper() == valor_filtro1.upper() and row[filtro_index3 - 1] and str(row[filtro_index3 - 1]).strip().upper() == valor_filtro3.upper()): datos_filtrados = {col: row[idx - 1] for col, idx in indices.items()} resumen_filtro1.append(datos_filtrados) contador1 += 1 # ---------- Filtro 2: OBSERVACION contiene URGENCIA VITAL + REMISION ---------- resumen_filtro2 = [] contador2 = 0 for row in ws.iter_rows(min_row=2, values_only=True): if (row[filtro_index2 - 1] and valor_filtro2 in str(row[filtro_index2 - 1]).upper() and row[filtro_index3 - 1] and str(row[filtro_index3 - 1]).strip().upper() == valor_filtro3.upper()): datos_filtrados = {col: row[idx - 1] for col, idx in indices.items()} resumen_filtro2.append(datos_filtrados) contador2 += 1 # ========== GENERAR EXCEL DE SALIDA ========== wb_out = openpyxl.Workbook() ws1 = wb_out.active ws1.title = "Tramite en proceso" ws2 = wb_out.create_sheet("Urgencia Vital") # Encabezados headers = list(indices.keys()) ws1.append(headers) ws2.append(headers) for reg in resumen_filtro1: ws1.append([reg.get(c) for c in headers]) for reg in resumen_filtro2: ws2.append([reg.get(c) for c in headers]) buffer_out = BytesIO() wb_out.save(buffer_out) resultado_bytes = buffer_out.getvalue() # ========== MOSTRAR RESULTADOS EN PANTALLA ========== stats_html = f"""
{contador1} Trámite en proceso (REMISIÓN)
{contador2} Urgencia Vital (REMISIÓN)
""" stats_el = document.getElementById("stats") stats_el.innerHTML = stats_html stats_el.style.display = "flex" # Vista previa en texto lineas = [] lineas.append("=== FILTRO 1: Trámite en proceso (solo REMISIÓN) ===") lineas.append(f"Total: {contador1}\n") for r in resumen_filtro1[:20]: lineas.append(str(r)) if contador1 > 20: lineas.append(f"... y {contador1 - 20} más (ver Excel descargado)") lineas.append("\n" + "=" * 55 + "\n") lineas.append("=== FILTRO 2: Observación contiene 'URGENCIA VITAL' (solo REMISIÓN) ===") lineas.append(f"Total: {contador2}\n") for r in resumen_filtro2[:20]: lineas.append(str(r)) if contador2 > 20: lineas.append(f"... y {contador2 - 20} más (ver Excel descargado)") set_output("\n".join(lineas)) set_status(f"✅ Procesado. {contador1 + contador2} registros encontrados en total.", "ok") document.getElementById("download-btn").style.display = "inline-block" except Exception as e: set_status(f"❌ Error: {e}", "error") set_output(str(e)) finally: mostrar_loading(False) # ---------- Descargar resultado ---------- def descargar_resultado(event=None): if resultado_bytes is None: return blob = window.Blob.new( [Uint8Array.new(resultado_bytes)], {"type": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"} ) url = window.URL.createObjectURL(blob) a = document.createElement("a") a.href = url a.download = "resultado_filtrado.xlsx" a.click() window.URL.revokeObjectURL(url)
